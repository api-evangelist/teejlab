    let get_all_user_info_url = "/edsn/knowledgebase/get_all_user_info"
    let curr_user_id = "None"
    console.log("test")

    $( document ).ready( function() {

        $('select[name="existing_agreements"]').select2({
            placeholder: "Select an agreement.",
        });
        $.ajax({
            method: "GET",
            url: get_all_user_info_url,
            contentType: "application/json",
            success: function (data) {
                let all_user_info_html = ""
                for(let user_single of data.all_user_info) {
                    if (user_single.id != curr_user_id) {
                        if (user_single.first_name && user_single.last_name) {
                            all_user_info_html += "<option value=" + user_single.id + (user_single.profile_image ? " title=" + user_single.profile_image : "") + ">" + user_single.first_name + " " + user_single.last_name + " (" + user_single.censored_email + ")</option>"
                        }
                        else {
                            all_user_info_html += "<option value=" + user_single.id + (user_single.profile_image ? " title=" + user_single.profile_image : "") + ">" + user_single.username + " (" + user_single.censored_email + ")</option>"

                        }
                    }
                }
                $('#user_id_api_mention').append(all_user_info_html)
                $('#user_id_api_mention2').append(all_user_info_html)
                select2_setup()
            }
        })
    });
</script>
